 * atomically promoted to their opaque storage key after completion.
 */

    readonly id = "local-disk";
    readonly kind: "local-disk";
    private readonly rootDirectory;
    private readonly uploadUrlBasePath;
    private readonly readUrlSecret;
    constructor(options: LocalDiskMediaStorageProviderOptions);
    health(): Promise<{
        status: "ok" | "degraded" | "down";
    }>;
    createUpload(input: CreateMediaStorageUploadInput): Promise<CreateMediaStorageUploadResult>;
    writeUpload(uploadId: string, body: AsyncIterable<Uint8Array>): Promise<void>;
    /** Compatibility alias for integrations that use the local provider directly. */
    stageUpload(uploadId: string, body: AsyncIterable<Uint8Array>): Promise<void>;
    discardUpload(uploadId: string): Promise<void>;
    completeUpload(input: CompleteMediaStorageUploadInput): Promise<MediaStoredObject>;
    createReadUrl(input: CreateMediaStorageReadUrlInput): Promise<{
        url: string;
        expiresAt: string;
    }>;

        expiresAtMs: number;
        signature: string;
